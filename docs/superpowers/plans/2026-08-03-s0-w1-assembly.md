// SecurityReport is the set of security subsystems an Orchestrator injects
// into every tool context (see the WithSandbox/WithNetworkPolicy/... block in

//
// This exists because Config is copied by value in New: a caller that assigns
// Config fields after calling New silently loses them, and the loss is
// invisible from outside the package. `yanshi doctor` reads Sandbox().Report()
// to describe the effective posture, and the bootstrap wiring test asserts
// every field is populated.
type SecurityReport struct {
	Sandbox       sandbox.Sandbox
	NetworkPolicy *netpolicy.Policy
	Approvals     *approval.Manager
	ShellManager  *shell.Manager
	SecureFactory secproc.Factory
}

// SecurityReport returns the security subsystems this Orchestrator holds.
func (o *Orchestrator) SecurityReport() SecurityReport {
	return SecurityReport{
		Sandbox:       o.sandbox,
		NetworkPolicy: o.networkPolicy,
		Approvals:     o.approvals,
		ShellManager:  o.shellManager,
		SecureFactory: o.secureFactory,
	}
}
